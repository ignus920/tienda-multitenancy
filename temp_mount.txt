    public function mount()
    {
        // Obtener viewType de la ruta o usar desktop por defecto
        $this->viewType = request()->route('viewType', 'desktop');
        $this->ensureTenantConnection();
        
        // Cargar productos desde la sesión (para modo edición)
        $this->quoterItems = session('quoter_items', []);
        
        // Cargar cliente seleccionado desde la sesión (para modo edición)
        $this->selectedCustomer = session('selected_customer', null);
        
        // Si hay un cliente seleccionado, también cargar su identificación en el campo de búsqueda
        if ($this->selectedCustomer && isset($this->selectedCustomer['identification'])) {
            $this->customerSearch = $this->selectedCustomer['identification'];
        }
        
        $this->calculateTotal();
    }
